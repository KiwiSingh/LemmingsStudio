cmake_minimum_required(VERSION 3.24)
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum supported macOS")
endif()
project(LemmingsPluginHost LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED JUCE_SOURCE)
  message(FATAL_ERROR "Set JUCE_SOURCE to the pinned JUCE 8.0.12 checkout")
endif()
add_subdirectory(${JUCE_SOURCE} JUCE)
add_library(lemmings_plugin_host SHARED Host.cpp)
target_compile_definitions(lemmings_plugin_host PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(lemmings_plugin_host PRIVATE juce::juce_audio_utils)
set_target_properties(lemmings_plugin_host PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(plugin_host_check Check.cpp)
target_link_libraries(plugin_host_check PRIVATE lemmings_plugin_host)

if(APPLE)
  target_link_options(lemmings_plugin_host PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
endif()
