cmake_minimum_required(VERSION 3.24)
project(LemmingsNative VERSION 0.1.0 LANGUAGES C CXX)
# Keep clean Windows VM deployment independent of a separate VC runtime install.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "")
include(FetchContent)
set(LEMMINGS_JUCE_DIR "" CACHE PATH "Existing pinned JUCE checkout")
if(LEMMINGS_JUCE_DIR)
  add_subdirectory("${LEMMINGS_JUCE_DIR}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG f72bad64d29715216226685810c5196bd0d79d77)
  FetchContent_MakeAvailable(JUCE)
endif()
add_library(studio_core STATIC src/Session.cpp src/Render.cpp)
target_include_directories(studio_core PUBLIC src)
target_compile_definitions(studio_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_ASIO=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(studio_core PUBLIC juce::juce_audio_utils juce::juce_dsp)
juce_add_gui_app(LemmingsNative PRODUCT_NAME "LemmingsStudio Native Preview" COMPANY_NAME "MagmaVance LLC" BUNDLE_ID com.magmavance.lemmings.nativepreview)
target_sources(LemmingsNative PRIVATE src/Main.cpp)
target_link_libraries(LemmingsNative PRIVATE studio_core)
set(amp_formats VST3 Standalone)
if(APPLE)
  list(APPEND amp_formats AU)
endif()
juce_add_plugin(LemmingAmp COMPANY_NAME "Kiwi Singh" BUNDLE_ID audio.lemmings.amp PLUGIN_MANUFACTURER_CODE Kwsn PLUGIN_CODE Lmap FORMATS ${amp_formats} PRODUCT_NAME "Lemming Amp" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(LemmingAmp PRIVATE src/AmpPlugin.cpp)
target_link_libraries(LemmingAmp PRIVATE studio_core)
juce_add_console_app(StudioChecks PRODUCT_NAME StudioChecks)
target_sources(StudioChecks PRIVATE tests/Checks.cpp)
target_link_libraries(StudioChecks PRIVATE studio_core)
enable_testing()
add_test(NAME native_session_and_audio COMMAND StudioChecks)
foreach(instrument IN ITEMS LemmingGuitar LemmingRockDrums)
  if(instrument STREQUAL "LemmingGuitar")
    set(instrument_code Lmgt)
    set(is_drums 0)
  else()
    set(instrument_code Lmdr)
    set(is_drums 1)
  endif()
  juce_add_plugin(${instrument} COMPANY_NAME "Kiwi Singh" BUNDLE_ID audio.lemmings.${instrument} PLUGIN_MANUFACTURER_CODE Kwsn PLUGIN_CODE ${instrument_code} FORMATS ${amp_formats} PRODUCT_NAME "${instrument} Preview" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(${instrument} PRIVATE src/SamplePlugin.cpp)
  target_compile_definitions(${instrument} PRIVATE LEMMINGS_DRUMS=${is_drums})
  target_link_libraries(${instrument} PRIVATE studio_core juce::juce_cryptography)
endforeach()
target_link_libraries(StudioChecks PRIVATE juce::juce_cryptography)

add_test(NAME native_plugin_wrappers COMMAND StudioChecks
  "${CMAKE_CURRENT_BINARY_DIR}/LemmingAmp_artefacts/$<CONFIG>/VST3/Lemming Amp.vst3"
  "${CMAKE_CURRENT_BINARY_DIR}/LemmingGuitar_artefacts/$<CONFIG>/VST3/LemmingGuitar Preview.vst3"
  "${CMAKE_CURRENT_BINARY_DIR}/LemmingRockDrums_artefacts/$<CONFIG>/VST3/LemmingRockDrums Preview.vst3")
